cmake_minimum_required(VERSION 3.20)
project(unpakz_native C CXX)

option(UNPAKZ_SANITIZE "Build everything with AddressSanitizer and UBSan (Linux/clang/gcc)" OFF)
if(UNPAKZ_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
  add_link_options(-fsanitize=address,undefined)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

# zlib-ng (zlib licence), vendored, built in zlib-compatible mode; libarchive uses it for gzip and ZIP deflate.
set(ZLIB_COMPAT ON CACHE BOOL "" FORCE)
set(ZLIB_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_GTEST OFF CACHE BOOL "" FORCE)
set(WITH_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(${CMAKE_SOURCE_DIR}/../third_party/zlib-ng zlib-ng EXCLUDE_FROM_ALL)
set(ZLIB_INCLUDE_DIR ${CMAKE_BINARY_DIR}/zlib-ng CACHE PATH "" FORCE)
set(ZLIB_LIBRARY zlibstatic CACHE STRING "" FORCE)
set(ZLIB_FOUND TRUE CACHE BOOL "" FORCE)

# libarchive, vendored read side only (see docs/decisions/0003, 0004 and docs/THIRD-PARTY.md). Only the library is built.
set(ENABLE_TAR OFF CACHE BOOL "" FORCE)
set(ENABLE_CPIO OFF CACHE BOOL "" FORCE)
set(ENABLE_CAT OFF CACHE BOOL "" FORCE)
set(ENABLE_UNZIP OFF CACHE BOOL "" FORCE)
set(ENABLE_TEST OFF CACHE BOOL "" FORCE)
set(ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(opt OPENSSL NETTLE MBEDTLS XML2 EXPAT LIBXML2 ICONV LZ4 LZO LIBB2 BZip2 LibGCC ZSTD LZMA CNG ACL XATTR)
  set(ENABLE_${opt} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(${CMAKE_SOURCE_DIR}/../third_party/libarchive libarchive EXCLUDE_FROM_ALL)

add_executable(smoke smoke.c)
target_link_libraries(smoke PRIVATE archive_static)
target_include_directories(smoke PRIVATE ${CMAKE_SOURCE_DIR}/../third_party/libarchive/libarchive)
target_compile_definitions(smoke PRIVATE LIBARCHIVE_STATIC)

# ---- our code ----
enable_testing()
set(UNPAKZ_SOURCES src/guard.cpp src/engine.cpp src/source.cpp src/fs_native.cpp src/volumes.cpp src/api.cpp src/entries.cpp src/libarchive_reader.cpp src/rar5.cpp)
add_library(unpakz_core STATIC ${UNPAKZ_SOURCES})
target_include_directories(unpakz_core PUBLIC src include)
target_link_libraries(unpakz_core PUBLIC archive_static zlibstatic)
target_include_directories(unpakz_core SYSTEM PUBLIC ${CMAKE_BINARY_DIR}/zlib-ng)
target_include_directories(unpakz_core SYSTEM PUBLIC ${CMAKE_SOURCE_DIR}/../third_party/libarchive/libarchive)
target_compile_definitions(unpakz_core PUBLIC LIBARCHIVE_STATIC)
if(MSVC)
  target_compile_options(unpakz_core PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(unpakz_core PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
endif()

add_executable(guard_test tests/guard_test.cpp)
target_link_libraries(guard_test PRIVATE unpakz_core)
add_test(NAME guard_test COMMAND guard_test)

add_executable(engine_test tests/engine_test.cpp)
target_link_libraries(engine_test PRIVATE unpakz_core)
add_test(NAME engine_test COMMAND engine_test)

foreach(t api_test volumes_test rar5_test rar5_container_test rar5_decoder_test)
  add_executable(${t} tests/${t}.cpp)
  target_link_libraries(${t} PRIVATE unpakz_core)
  add_test(NAME ${t} COMMAND ${t})
endforeach()
target_compile_definitions(rar5_test PRIVATE UNPAKZ_FIXTURE_DIR="${CMAKE_SOURCE_DIR}/../tests/fixtures")

# The shipped library: C API only, static CRT so users need no redistributable.
add_library(unpakz SHARED ${UNPAKZ_SOURCES})
target_compile_definitions(unpakz PRIVATE UNPAKZ_BUILD_DLL LIBARCHIVE_STATIC)
target_include_directories(unpakz PUBLIC src include)
target_include_directories(unpakz SYSTEM PRIVATE ${CMAKE_SOURCE_DIR}/../third_party/libarchive/libarchive)
target_link_libraries(unpakz PRIVATE archive_static zlibstatic)
target_include_directories(unpakz SYSTEM PRIVATE ${CMAKE_BINARY_DIR}/zlib-ng)
# The file is unpakz_native.dll, not unpakz.dll: on a case-insensitive file system "unpakz.dll" is also what the loader looks for when it needs the managed assembly "Unpakz" (the app).
set_target_properties(unpakz PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON OUTPUT_NAME unpakz_native)
if(MSVC)
  target_compile_options(unpakz PRIVATE /W4 /WX /permissive- /utf-8)
else()
  target_compile_options(unpakz PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
endif()

# Test driver over the shared library's C API (not shipped).
add_executable(unpakz_testcli tools/unpakz_testcli.cpp src/guard.cpp)
target_link_libraries(unpakz_testcli PRIVATE unpakz)
